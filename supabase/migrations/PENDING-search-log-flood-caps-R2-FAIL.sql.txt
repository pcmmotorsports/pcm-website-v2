-- ── 2. 函式:上限改成原子計數 ─────────────────────────────────────────────
-- 🛑 **本體其餘部分逐字不動**(遮 email / 兩端截斷 / ON CONFLICT 那三段照抄)。
CREATE OR REPLACE FUNCTION public.log_search_query(
  p_query_raw    text,
  p_path         text,
  p_unmatched    text    DEFAULT NULL,
  p_result_count integer DEFAULT NULL
) RETURNS void
LANGUAGE sql
SECURITY DEFINER
SET search_path = ''
AS $fn$
  -- 🔴🔴 **原子計數**:`DO UPDATE … WHERE c.n < 5000` —— 撞到上限時**不回傳列**,
  --    下面那個 `INSERT … SELECT … FROM bump` 就自然變成零列 ⇒ **一列都不寫**。
  --    🎯 一次索引更新, 不掃表;並發由那一列的 row lock 序列化 ⇒ **是硬上限, 不是近似。**
  --    🛑 **代價明寫**:撞到上限的那一小時**語料是缺的**, 而**沒有東西會告訴我們它缺了**
  --       ⇒ 要知道有沒有撞到, 去看 `search_log_hour_counter` 那一列的 `n` 是不是剛好 5000。
  WITH bump AS (
    INSERT INTO public.search_log_hour_counter AS c (hour_bucket, n)
    VALUES (pg_catalog.date_trunc('hour', pg_catalog.now()), 1)
    ON CONFLICT (hour_bucket) DO UPDATE SET n = c.n + 1 WHERE c.n < 5000
    RETURNING 1 AS ok
  )
  INSERT INTO public.search_queries (query_raw, path, unmatched, result_count)
  SELECT
    -- 🔵 兩端都截:CHECK 會擋掉過長的, 而**擋掉 = 那一筆語料消失**
    --    ⇒ 這裡先截成合法的, 讓它進得去。截斷長度與 CHECK 一致(100)。
    -- 🔴🔴 **遮掉 email(2026-09-04 codex must-fix ②)** —— 而它做在【DB 這一層】,
    --    不是做在 TS:呼叫端可能不只一個, 而遮罩漏在任何一個呼叫端上都沒有東西會叫。
    pg_catalog.left(pg_catalog.btrim(public.mask_obvious_pii(p_query_raw)), 100),
    p_path,
    pg_catalog.left(public.mask_obvious_pii(p_unmatched), 100),
    p_result_count
  FROM bump
  -- 🔴 撞到那道 UNIQUE 就靜靜跳過 —— **不得讓它變成一個 error**:
  --    呼叫端是 fire-and-forget, 而「同一小時第二次搜同一個字」是**完全正常的客人行為**,
  --    讓它進 log 會把真的失敗淹掉。
  ON CONFLICT (query_raw, path, created_at) DO NOTHING;
$fn$;

-- ── 3. purge 函式(分批, 有界)────────────────────────────────────────────
-- 🔴🔴 **不分批跑迴圈, 改成【每一次只做有界的一小塊, 排程跑得勤】** ——
--    而那是**被 PostgreSQL 的規則逼出來的, 不是我偷懶**(2026-09-05 本機拋棄式 PG 17.10 實測):
--      `CREATE PROCEDURE … SECURITY DEFINER          ⇒ COMMIT ⇒ invalid transaction termination`
--      `CREATE PROCEDURE … SET search_path=''        ⇒ COMMIT ⇒ invalid transaction termination`
--      🟢 正對照:兩個子句都不加的最小 procedure ⇒ `COMMIT` **成功**(尺是活的)
--    ⇒ 🎯 **本 repo 對這種函式【兩個子句都是硬性要求】⇒ 交易控制在這裡結構上不可能。**
--    ⇒ ⛔ ~~R2 第一版寫成 procedure + 迴圈裡 COMMIT~~ —— **它連跑都跑不起來**, 而
--      **靜態檢查與「乾淨貼 rc=0」兩關【都是綠的】** ⇒ 📌 抓到它的是**真的去 CALL 了一次**。
--    ✅ 改法:**兩半【各自】一批 ≤ 5,000**, 排程改成**每 10 分鐘**一次 ⇒
--      單次交易小、WAL 有界、不用交易控制;積欠會在後續幾發自己收斂。
--    ⛔ ~~「一次呼叫只刪一批 ≤5,000」~~ **那句是假的**(codex R2 第 6 條):
--      過期與超量是**兩個 DELETE** ⇒ **單發最多 10,000 列**;計數表那一發原本**還沒有上界**(已補 LIMIT)。
--      🔵 排乾能力 = 10,000 × 6 × 24 = **144 萬列/天**(⚠️ 而**排程延遲沒算進去**)。
--    ⚠️ **代價明寫**:一次大爆發之後**不會立刻清乾淨** —— 它是**漸進收斂**, 不是一次到位。
CREATE FUNCTION public.purge_search_queries()
RETURNS TABLE (deleted_expired bigint, deleted_overflow bigint, more_to_do boolean)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $fn$
DECLARE
  -- 🔴 保留期:Sean 2026-08-21 拍「搜尋紀錄保存 2 年」。
  c_retention CONSTANT interval := interval '2 years';
  -- 🔴 總量上限:Sean 2026-09-05 拍甲。改值只要改這一行。
  c_max_rows  CONSTANT bigint  := 2000000;
  -- 🔴 單次上界 —— 這就是「有界」那一格。
  c_batch     CONSTANT integer := 5000;
  v_expired  bigint := 0;
  v_overflow bigint := 0;
BEGIN
  -- ① 過期那半(單批)
  DELETE FROM public.search_queries
   WHERE id IN (SELECT s.id FROM public.search_queries s
                 WHERE s.created_at < pg_catalog.now() - c_retention
                 LIMIT c_batch);
  GET DIAGNOSTICS v_expired = ROW_COUNT;

  -- ②a 🔴🔴 **前置:表【小到不可能有 c_max_rows 列】就整段跳過**
  --     (2026-09-05 折 codex R2 第 5 條 must-fix;主視窗 `-f8` 裁「只折這一條」)
  --
  --  🔴 **它折的是什麼**:下面 ②b 那句 `ORDER BY id DESC OFFSET c_max_rows` **每一發都要跨過
  --     200 萬個 index item**;144 發/日 ⇒ **每日約 2.88 億**。而 codex 逐字:防線自己變成常態成本。
  --  🎯 **而那句話只有在【真的超量】時才需要跑** ⇒ 加一道 O(1) 的前置, 沒超量就不跑
  --     ⇒ **在正常世界(也就是每一天, 直到真的被灌爆)那個放大器根本不執行。**
  --  ✅ **⇒ 這一格不需要任何壓測就成立** —— 它把「量它貴不貴」換成「讓它不跑」。
  --
  --  ⛔ ~~我第一版要用 `pg_class.reltuples`~~ —— 🔴 **它在我寫下去之前就被量掉了**:
  --     2026-09-05 唯讀實查正式庫 ⇒ `search_queries` 的 `reltuples` = **-1**,
  --     `last_analyze` 與 `last_autoanalyze` **都是 NULL**(從未被 ANALYZE 過)。
  --     🛑 **`-1` 是 PostgreSQL 的「不知道」哨兵值** ⇒ 而 `-1 < 2000000` 為真
  --     ⇒ 那道前置會**跳過 purge**, 也就是**把「沒有資訊」讀成「表很小」**。
  --     🟢 正對照:同一發問 `products` ⇒ `reltuples` = 25038(尺會動, 不是它壞了)。
  --
  --  ✅ **改用 `pg_relation_size`, 而它是【可證明】的**:
  --     · 它**永遠精確**、不依賴 ANALYZE(它問的是檔案佔幾個 page)
  --     · **大小是列數的上界**:heap tuple header MAXALIGN 後 24 bytes + item pointer 4 bytes
  --       ⇒ **任何一列至少 28 bytes**;本表還有 `id`(int8)與 `created_at`(timestamptz)
  --       ⇒ 至少 40。**取 32 是刻意保守的下界。**
  --     ⇒ 📌 **表小於 `c_max_rows × 32` ⇒ 不可能有 c_max_rows 列。這是算出來的, 不是量出來的。**
  --  🔵 **而 bloat / dead tuple 只會讓表【變大】** ⇒ 它推向「不跳過」= 照樣跑那句
  --     ⇒ **失效方向是安全的那一側**(多做一次, 不是少做)。
  --  🔵 **TOAST 也不破**:長字串搬去 TOAST 之後主表那一列仍留一個約 18 bytes 的指標
  --     ⇒ 每列仍 ≥ 28 ⇒ 32 這個下界照樣成立。
  --
--  🔬 **拋棄式 PG 17.10 實跑(2026-09-05;真的函式本體, 從本檔抽出來載入, 沒有手抄)**:
--     ```
--     每列實際 bytes  52.18(query_raw 只放 1 個字元 = 最短的情況)
--                     ⇒ 我宣稱的下界 32 成立而且有餘裕
--     門檻換算        64,000,000 bytes ÷ 52.18 ≈ 【122.7 萬列】就會開閘
--                     ⇒ 而上限是 200 萬 ⇒ 📌 **閘【比需要的時候更早開】, 它不可能在超量時擋住**
--     世界 A  20 萬列 / 10,436,608 bytes ⇒ 閘 f ⇒ deleted_overflow = 0, 零錯誤
--     世界 B 205 萬列 / 106,971,136 bytes ⇒ 閘 t ⇒ deleted_overflow = 5000, more_to_do = t
--     ```
--  🔴🔴 **而有一件我【沒有證明】, 照實寫**:世界 A 的「0」與「跑了但沒東西刪」**在結果上長一樣**。
--     我試著用 `pg_stat_user_tables.idx_scan` 分開它們 —— 🛑 **而正對照(手動跑同一句)也回 0**
--     ⇒ **那把尺沒動, 兩個讀數都作廢, 我不拿它當證據。**
--     ⇒ 📌 **「它沒跑」這件事我靠的是 `IF` 的語意 + 上面量到的閘值(A 是 f、B 是 t)**,
--        **不是一個獨立的觀察。** 而 `IF` 為假時不執行內層是語言保證, 不是本片的宣稱。
--
--  🛑 **它擋不住什麼**:①這一格**只跳過 ②b**, 過期那半(①)照跑 —— 那是對的, 它便宜
  --     ②`c_max_rows` 或表結構若改, **這個下界要重算**(改值的人請重讀本段)
  --     ③它**不保證**超量時那句就便宜 —— 只保證**沒超量時它不跑**。
  IF pg_catalog.pg_relation_size('public.search_queries'::regclass) >= c_max_rows * 32 THEN

  -- ②b 總量那半(單批)—— 🔴 只刪【超出上限的最舊那些】。
  --    用 id 排序而不是 created_at:`created_at` 被截到整點
  --    ⇒ **同一小時的列它分不出先後**, 而 `id` 是 IDENTITY, 嚴格遞增。
  DELETE FROM public.search_queries
   WHERE id IN (SELECT s.id FROM public.search_queries s
                 ORDER BY s.id DESC OFFSET c_max_rows LIMIT c_batch);
  GET DIAGNOSTICS v_overflow = ROW_COUNT;
  END IF;   -- ②a 的前置:表太小 ⇒ v_overflow 維持 0, 而那是【正確的 0】不是【沒跑的 0】

  -- ③ 計數表也要清 —— 一小時一列, 一年 8,760 列, 小而**不是零**。
  DELETE FROM public.search_log_hour_counter
   WHERE hour_bucket IN (SELECT c.hour_bucket FROM public.search_log_hour_counter c
                          WHERE c.hour_bucket < pg_catalog.now() - interval '90 days'
                          LIMIT c_batch);

  -- ④ 🔴🔴 **心跳** —— 少了它, 這支排程死掉**沒有任何人會知道**(codex 第 10 條)。
  INSERT INTO public.sweeper_heartbeat AS h (job_name, last_success_at, consecutive_failures, updated_at)
  VALUES ('pcm-search-log-purge', pg_catalog.now(), 0, pg_catalog.now())
  ON CONFLICT (job_name) DO UPDATE
    SET last_success_at = pg_catalog.now(), consecutive_failures = 0, updated_at = pg_catalog.now();

  -- 🔵 兩半任一批「剛好滿」⇒ 很可能還有 ⇒ 下一發再來。**這是提示, 不是保證。**
  RETURN QUERY SELECT v_expired, v_overflow,
                      (v_expired >= c_batch OR v_overflow >= c_batch);
END;
$fn$;

-- 🔴 函式**出生就對 PUBLIC 開 EXECUTE** ⇒ 先收再給。
-- 🛑 這一支**誰都不給** —— 它只由 `pg_cron`(postgres 身分)呼叫。
REVOKE ALL ON FUNCTION public.purge_search_queries() FROM PUBLIC;
REVOKE ALL ON FUNCTION public.purge_search_queries() FROM authenticated;
REVOKE ALL ON FUNCTION public.purge_search_queries() FROM anon;
REVOKE ALL ON FUNCTION public.purge_search_queries() FROM service_role;

-- ── 4. 排程(形狀逐格抄 `20260809170000:70-111` —— 🔴 那支就是【純 SQL job 的先例】)──
-- ⛔ ~~R1 檔頭寫「既有六支都走 invoke_cron_route ⇒ 本支偏離」~~ ⇒ **那句是假的**(codex 第 8 條):
--    `pcm-expire-unpaid-orders` 就是純 SQL。📌 **我為一個不存在的偏離寫了一整段辯護。**
DO $$
DECLARE v_id bigint;
BEGIN
  v_id := cron.schedule('pcm-search-log-purge', '*/10 * * * *',
    $job$SELECT public.purge_search_queries()$job$);
  PERFORM cron.alter_job(job_id => v_id, active => true);  -- by-name upsert 不改 active, 顯式設 true
END $$;

-- ── 5. 事後閘(fail-closed:任一格不成立 ⇒ 整檔 ROLLBACK)──────────────────
DO $$
DECLARE v_cnt int; v_ok boolean;
BEGIN
  -- ① 原子計數那段真的進去了 —— 🔴 **釘的是 predicate, 不是註解**(codex 第 11 條:
  --    R1 只找 `%< 5000%`, 而註解留著、predicate 拿掉也會綠)。
  SELECT pg_catalog.count(*) INTO v_cnt
    FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
   WHERE n.nspname = 'public' AND p.proname = 'log_search_query'
     AND p.prosrc LIKE '%ON CONFLICT (hour_bucket) DO UPDATE SET n = c.n + 1 WHERE c.n < 5000%';
  IF v_cnt <> 1 THEN RAISE EXCEPTION '事後閘①:log_search_query 裡找不到原子計數那一句'; END IF;

  -- ①b 🔴 **②a 那道前置真的進去了** —— 釘的是 **predicate**, 不是註解
  --     (同 codex 第 11 條的教訓:只找註解的話, 註解留著而 IF 拿掉也會綠)。
  --     🛑 而它**證不到那道前置是對的** —— 只證「那一句在函式體裡」。
  SELECT pg_catalog.count(*) INTO v_cnt
    FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
   WHERE n.nspname = 'public' AND p.proname = 'purge_search_queries'
     AND p.prosrc LIKE '%pg_relation_size(''public.search_queries''::regclass) >= c_max_rows * 32%';
  IF v_cnt <> 1 THEN
    RAISE EXCEPTION '事後閘①b:purge_search_queries 裡找不到 ②a 的大小前置(那句 IF)';
  END IF;

  -- ② 計數表在, 而且 anon/authenticated/service_role 一項權限都沒有
  IF pg_catalog.to_regclass('public.search_log_hour_counter') IS NULL THEN
    RAISE EXCEPTION '事後閘②a:計數表沒建成';
  END IF;
  FOR v_ok IN
    SELECT pg_catalog.has_table_privilege(r, 'public.search_log_hour_counter', p)
      FROM unnest(ARRAY['anon','authenticated','service_role']) r,
           unnest(ARRAY['SELECT','INSERT','UPDATE','DELETE','TRUNCATE','REFERENCES','TRIGGER','MAINTAIN']) p
  LOOP
    IF v_ok THEN RAISE EXCEPTION '事後閘②b:計數表上還有角色有權限'; END IF;
  END LOOP;

  -- ③ purge:存在 / DEFINER / search_path 鎖空 / owner —— 🔴 **三個世界分三句**
  --    (R1 把它們寫成同一句, 而那正是我今晚一直在抓的病)。
  IF NOT EXISTS (SELECT 1 FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
                  WHERE n.nspname='public' AND p.proname='purge_search_queries') THEN
    RAISE EXCEPTION '事後閘③a:purge_search_queries 根本沒建成';
  END IF;
  SELECT p.prosecdef INTO v_ok FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
   WHERE n.nspname='public' AND p.proname='purge_search_queries';
  IF v_ok IS NOT TRUE THEN RAISE EXCEPTION '事後閘③b:purge 不是 SECURITY DEFINER'; END IF;
  -- 🔴 存進去的字面是 `search_path=""`(帶兩個引號)—— R1 比 `search_path=` ⇒ 對正確的函式**誤報**。
  --    那是**拋棄式 PG 當場抓到的**, 不是讀文件推的。
  SELECT p.proconfig @> ARRAY['search_path=""'] INTO v_ok
    FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
   WHERE n.nspname='public' AND p.proname='purge_search_queries';
  IF v_ok IS NOT TRUE THEN RAISE EXCEPTION '事後閘③c:purge 的 search_path 沒鎖成空'; END IF;
  -- 🔴 **它必須【不含】交易控制** —— 帶 SECURITY DEFINER 或 SET 的函式/程序用 COMMIT 會當場報錯,
  --    而那個錯【只有真的呼叫才看得到】:靜態檢查與「乾淨貼 rc=0」兩關都是綠的。
  -- ⚠️ **這道閘很弱, 明寫**(codex R2 第 8 條):它只擋一種字面 ——
  --    `commit ;`、大小寫、`ROLLBACK`、動態 SQL、間接 `CALL` **它一個都擋不到**。
  --    ✅ 而**真正擋住這件事的不是它, 是 PostgreSQL 自己**:帶 DEFINER/SET 的函式跑到交易控制會當場報錯。
  --    ⇒ 📌 本格的用途只是「讓下一個人在【貼之前】就撞到」, 不是保證。
  SELECT p.prosrc !~* '\m(commit|rollback)\M' INTO v_ok
    FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
   WHERE n.nspname='public' AND p.proname='purge_search_queries';
  IF v_ok IS NOT TRUE THEN RAISE EXCEPTION '事後閘③d:purge 裡有 COMMIT ⇒ 它在 DEFINER + SET 之下跑不起來'; END IF;
  SELECT p.proowner = 'postgres'::regrole INTO v_ok
    FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
   WHERE n.nspname='public' AND p.proname='purge_search_queries';
  IF v_ok IS NOT TRUE THEN RAISE EXCEPTION '事後閘③e:purge 的 owner 不是 postgres'; END IF;

  -- ④ purge 除了 owner 之外一個 grantee 都沒有(形狀抄先例 `:52-58`)
  IF EXISTS (
    SELECT 1 FROM pg_proc p, LATERAL aclexplode(coalesce(p.proacl, acldefault('f', p.proowner))) a
     WHERE p.oid = 'public.purge_search_queries()'::regprocedure AND a.grantee <> p.proowner
  ) THEN
    RAISE EXCEPTION '事後閘④:purge 有 owner 以外的 grantee';
  END IF;

  -- ⑤ 🔵 **反向對照** —— 少了這一格, 「全部 REVOKE 掉」與「本來就沒開過」印同一個綠。
  --    ⚠️ **它守得住什麼、守不住什麼(codex 第 12 條)**:它只證「`anon` 有【有效】EXECUTE」,
  --      權限若來自 PUBLIC 或角色繼承也會綠 ⇒ **守得住「客人完全寫不進去」, 守不住「只由預期的直接 grant 開放」。**
  --      ⇒ 所以下面再加一句:`authenticated` / `service_role` 必須**還是**收掉的。
  IF NOT pg_catalog.has_function_privilege(
           'anon', 'public.log_search_query(text, text, text, integer)', 'EXECUTE') THEN
    RAISE EXCEPTION '事後閘⑤a:anon 對 log_search_query 的 EXECUTE 不見了 ⇒ 客人端會靜靜地全部寫不進去';
  END IF;
  IF pg_catalog.has_function_privilege('authenticated', 'public.log_search_query(text, text, text, integer)', 'EXECUTE')
     OR pg_catalog.has_function_privilege('service_role', 'public.log_search_query(text, text, text, integer)', 'EXECUTE') THEN
    RAISE EXCEPTION '事後閘⑤b:log_search_query 對 authenticated/service_role 的 EXECUTE 跑出來了';
  END IF;

  -- ⑥ 排程:先數同名【全部】(併 username 進 WHERE 會濾掉跨角色的第二筆 ⇒ 看不見的重複排程)
  SELECT pg_catalog.count(*) INTO v_cnt FROM cron.job WHERE jobname = 'pcm-search-log-purge';
  IF v_cnt <> 1 THEN RAISE EXCEPTION 'LOGFLOOD:同名 job 不是恰一筆(實 %, 可能跨角色重複);拒繼續', v_cnt; END IF;
  SELECT pg_catalog.count(*) INTO v_cnt FROM cron.job
   WHERE jobname = 'pcm-search-log-purge' AND active
     AND database = current_database()
     AND schedule = '*/10 * * * *'
     AND command  = 'SELECT public.purge_search_queries()'
     AND username = 'postgres';
  IF v_cnt <> 1 THEN RAISE EXCEPTION 'LOGFLOOD:job 不是恰一筆 active 且四個欄位字面相符(實 %)', v_cnt; END IF;
END $$;

-- ⚠️ **誠實邊界(抄先例 `:74-78`)**:上面只證「`cron.job` 這一列的設定是對的」。
--    scheduler 沒在跑 / `cron.launch_active_jobs=off` / 首次執行就失敗 / run logging 被關掉 ——
--    **這四種本斷言全部照樣通過。** 真正的「它有在跑」要看 `cron.job_run_details`,
--    以及 `sweeper_heartbeat` 那一列的 `last_success_at` 有沒有在動。

-- 新物件收權斷言(樣板逐字抄自 `20260904240000:135-228`, 只換清單 —— 🔴 不重寫, 重寫會漂)
DO $newobj_guard$
DECLARE
  -- 🔴 結尾的 ::text[] 不能拿掉 —— 清單清空時 ARRAY[] 無法推斷型別。
  v_relations text[] := ARRAY[
    'public.search_log_hour_counter'
  ]::text[];
  v_functions text[] := ARRAY[
    'public.purge_search_queries()'
  ]::text[];
  v_declares_nothing boolean := false;

  r          text;
  v_oid      oid;
  v_bad      int := 0;
  v_first    text;
  v_checked  int := 0;
  v_priv     text;
  v_col      text;
BEGIN
  IF cardinality(v_relations) = 0 AND cardinality(v_functions) = 0 THEN
    IF NOT v_declares_nothing THEN
      RAISE EXCEPTION
        '新物件收權斷言:兩份清單都是空的。本檔若真的沒新建物件，請把 v_declares_nothing 設成 true（明示），不要留空。';
    END IF;
    RAISE NOTICE '新物件收權斷言:本檔明示未新建任何物件，略過（已留痕）。';
    RETURN;
  END IF;

  FOREACH r IN ARRAY v_relations LOOP
    v_oid := pg_catalog.to_regclass(r);
    IF v_oid IS NULL THEN
      RAISE EXCEPTION '新物件收權斷言:找不到關聯 % —— 名字打錯或沒建成。拒繼續。', r;
    END IF;
    v_checked := v_checked + 1;
    -- 🔴🔴 **權限清單由伺服器推導,不手寫**(codex 2026-08-17 `must-fix`)。
    --    E-684 樣板手寫七種,而 **PG 17 有八種 —— 少了 `MAINTAIN`**。
    --    而本檔自己的病構造輸出就印著 `anon=MAINTAIN` ⇒ **樣板掃不到它自己舉的那個例子。**
    --    改成從 `acldefault('r', owner)` 推導:PG 之後再加第九種,這一臂自動入列。
    --    📎 同一個修法 B1-b 已經走過(`docs/specs/2026-08-16-m4b-e8b-b1b-migration-draft.sql:343-346`)。
    --    🔴 `DISTINCT` 在這裡是承重的:這是迴圈,同一權限型別出現兩次會讓 v_bad 多加一次。
    FOR v_priv IN
      SELECT DISTINCT d.privilege_type
        FROM aclexplode(acldefault('r', (SELECT relowner FROM pg_class WHERE oid = v_oid))) d
    LOOP
      IF has_table_privilege('anon', v_oid, v_priv)
         OR has_table_privilege('authenticated', v_oid, v_priv) THEN
        v_bad := v_bad + 1;
        IF v_first IS NULL THEN v_first := format('%s 上仍有 %s', r, v_priv); END IF;
      END IF;
    END LOOP;

    -- 🔴🔴 欄級授權必須另外問 —— `has_table_privilege` 對【只有欄級授權】的情況回 false。
    FOR v_col IN
      SELECT a.attname FROM pg_attribute a
       WHERE a.attrelid = v_oid AND a.attnum > 0 AND NOT a.attisdropped
    LOOP
      FOREACH v_priv IN ARRAY ARRAY['SELECT','INSERT','UPDATE','REFERENCES'] LOOP
        IF has_column_privilege('anon', v_oid, v_col, v_priv)
           OR has_column_privilege('authenticated', v_oid, v_col, v_priv) THEN
          v_bad := v_bad + 1;
          IF v_first IS NULL THEN
            v_first := format('%s.%s 上仍有【欄級】%s', r, v_col, v_priv);
          END IF;
        END IF;
      END LOOP;
    END LOOP;
  END LOOP;

  FOREACH r IN ARRAY v_functions LOOP
    v_oid := pg_catalog.to_regprocedure(r);
    IF v_oid IS NULL THEN
      RAISE EXCEPTION '新物件收權斷言:找不到函式 % —— 簽名打錯或沒建成。拒繼續。', r;
    END IF;
    v_checked := v_checked + 1;
    IF pg_catalog.has_function_privilege('anon', v_oid, 'EXECUTE')
       OR pg_catalog.has_function_privilege('authenticated', v_oid, 'EXECUTE') THEN
      v_bad := v_bad + 1;
      IF v_first IS NULL THEN v_first := format('%s 上仍有 EXECUTE', r); END IF;
    END IF;
  END LOOP;

  IF v_checked = 0 THEN
    RAISE EXCEPTION '新物件收權斷言:檢查數為 0 —— 這個斷言沒有分母，不算通過。';
  END IF;

  IF v_bad > 0 THEN
    RAISE EXCEPTION
      E'❌ 新物件收權斷言失敗:anon/authenticated 仍持有 % 項權限（第一個:%）。檢查了 % 個物件。\n'
      '   ⇒ 補上:REVOKE ALL ON <物件> FROM PUBLIC, anon, authenticated;\n'
      '   🔴 兩道都要下:FROM PUBLIC 收不到具名授權，FROM 具名 收不到 PUBLIC 授權。',
      v_bad, v_first, v_checked;
  END IF;

  RAISE NOTICE '✅ 新物件收權斷言通過:檢查 % 個物件，anon/authenticated 權限 0 項。', v_checked;
END
$newobj_guard$;

COMMIT;
