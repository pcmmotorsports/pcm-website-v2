## 1 外鍵分母 —— 問 `pg_constraint`,不猜

```
指到 public.customers(user_id) 的外鍵【七條】:
  🛑 ON DELETE RESTRICT(會擋住刪除)
     orders.customer_user_id · shipments.customer_user_id · coupon_redemptions.user_id
  🔴 ON DELETE CASCADE(不會擋,會【跟著一起被刪掉】)
     customer_addresses · customer_vehicles · customer_wallet_ledger · customer_favorites
```

🔴🔴 **CASCADE 那四條是這一份最該讀的一段:**
📌 **`RESTRICT` 會讓你知道你踩到東西;`CASCADE` 不會 —— 它會安靜地把它們一起帶走。**
🛑 **而 `customer_wallet_ledger` 在那四條裡面** —— 那是**儲值金的流水帳**。今天那三個帳號的錢包都是空的(見 §2),
　**而這句話的射程只到今天**:誰在刪之前往測試帳號充了值,那筆流水會**無聲消失**。

## 5 🔴 刪的 SQL(**一發都沒有跑**)

```sql
-- 🛑 只刪【兩個】:g3-preview-sandbox 與 uitest。
--    g3-sandbox-test 不在裡面 —— 它名下 3 張訂單 + 1 個箱子(含 CH6D75 新竹首箱)。
-- 🔴 用 user_id 不用 email —— email 改得動, uuid 改不動。
BEGIN;

DO $gate$
DECLARE
  v_ids uuid[] := ARRAY['d5e621ed-8632-49c0-80ff-039ea730a385',
                        'cd06eb48-d41b-4158-9816-23b436a264df']::uuid[];
  v_id  uuid;
  v_n   bigint;
BEGIN
  FOREACH v_id IN ARRAY v_ids LOOP
    IF NOT EXISTS (SELECT 1 FROM public.customers c WHERE c.user_id = v_id) THEN
      RAISE EXCEPTION '前置閘:找不到 % ⇒ 它已經被刪過, 或 uuid 打錯了。拒繼續', v_id;
    END IF;

    SELECT count(*) INTO v_n FROM public.orders o WHERE o.customer_user_id = v_id;
    IF v_n <> 0 THEN
      RAISE EXCEPTION '前置閘:% 名下有 % 張訂單 ⇒ 拒刪', v_id, v_n;
    END IF;

    SELECT count(*) INTO v_n FROM public.shipments s WHERE s.customer_user_id = v_id;
    IF v_n <> 0 THEN
      RAISE EXCEPTION '前置閘:% 名下有 % 個箱子 ⇒ 拒刪', v_id, v_n;
    END IF;

    SELECT count(*) INTO v_n FROM public.coupon_redemptions r WHERE r.user_id = v_id;
    IF v_n <> 0 THEN
      RAISE EXCEPTION '前置閘:% 名下有 % 筆折價券核銷 ⇒ 拒刪', v_id, v_n;
    END IF;

    -- 🔴 CASCADE 那四條【不會擋】⇒ 這裡把它們【印出來】, 不讓它們無聲消失
    SELECT count(*) INTO v_n FROM public.customer_wallet_ledger w WHERE w.customer_user_id = v_id;
    IF v_n <> 0 THEN
      RAISE EXCEPTION '前置閘:% 有 % 筆儲值金流水 ⇒ 那是錢, 拒刪(要人先看過)', v_id, v_n;
    END IF;

    SELECT count(*) INTO v_n FROM public.customer_addresses a WHERE a.customer_user_id = v_id;
    RAISE NOTICE '⚠️ % 會連帶刪掉 % 筆地址(ON DELETE CASCADE)', v_id, v_n;
    SELECT count(*) INTO v_n FROM public.customer_vehicles v WHERE v.customer_user_id = v_id;
    RAISE NOTICE '⚠️ % 會連帶刪掉 % 筆愛車(ON DELETE CASCADE)', v_id, v_n;
    SELECT count(*) INTO v_n FROM public.customer_favorites f WHERE f.customer_user_id = v_id;
    RAISE NOTICE '⚠️ % 會連帶刪掉 % 筆收藏(ON DELETE CASCADE)', v_id, v_n;
  END LOOP;
END
$gate$;

DELETE FROM public.customers
 WHERE user_id IN ('d5e621ed-8632-49c0-80ff-039ea730a385',
                   'cd06eb48-d41b-4158-9816-23b436a264df');

DO $assert$
DECLARE v_total bigint; v_left bigint;
BEGIN
  SELECT count(*) INTO v_total FROM public.customers;
  IF v_total <> 13 THEN
    RAISE EXCEPTION '事後斷言:customers 應為 13(刪前 15), 實際 % ⇒ 拒繼續', v_total;
  END IF;

  SELECT count(*) INTO v_left FROM public.customers
   WHERE email IN ('g3-preview-sandbox@pcmmotorsports.com','uitest@pcmmotorsports.com');
  IF v_left <> 0 THEN
    RAISE EXCEPTION '事後斷言:那兩個信箱還在(剩 % 列)⇒ 拒繼續', v_left;
  END IF;

  -- 🔴 反向斷言:不准順手刪到第三個
  IF NOT EXISTS (SELECT 1 FROM public.customers
                  WHERE user_id = '8d402365-5740-4c77-ac4a-1f5eb68f643b'::uuid) THEN
    RAISE EXCEPTION '事後斷言:g3-sandbox-test 不見了 ⇒ 它不該被這一發碰到, 拒繼續';
  END IF;

  RAISE NOTICE '✅ 刪掉 2 個。🔴 而 auth.users 那一側【沒有被碰到】—— 那兩個信箱仍然登得進來, 要去 Dashboard 的 Authentication 另外刪。';
END
$assert$;

COMMIT;
```
